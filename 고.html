import pygame import random import time import math # --- 게임 초기화 --- pygame.init() # 모바일 세로 모드 비율 (예: 400x700) WIDTH, HEIGHT = 400, 700 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("좀비 아포칼립스 무한 열차 - 모바일 프로토타입") try: font = pygame.font.SysFont("malgun gothic", 18) except: font = pygame.font.SysFont("apple gothic", 18) # --- 색상 설정 --- WHITE, BLACK = (255, 255, 255), (0, 0, 0) TRAIN_COLOR = (120, 120, 120) CONTROL_PANEL_COLOR = (50, 150, 255) PLAYER_COLOR = (255, 50, 50) BG_COLOR = (40, 60, 40) TRACK_COLOR = (80, 80, 80) TIE_COLOR = (139, 69, 19) STATION_COLOR = (200, 150, 50) UI_COLOR = (200, 200, 200, 150) # 반투명 UI # --- 가상 조이스틱 클래스 --- class VirtualJoystick: def __init__(self, x, y, radius): self.base_pos = pygame.math.Vector2(x, y) self.stick_pos = pygame.math.Vector2(x, y) self.radius = radius self.active = False def update(self, mouse_pos, mouse_pressed): mouse_vec = pygame.math.Vector2(mouse_pos) # 화면 터치 중일 때 if mouse_pressed[0]: # 조이스틱 영역을 처음 터치했을 때 활성화 if not self.active and self.base_pos.distance_to(mouse_vec) <= self.radius * 1.5: self.active = True if self.active: self.stick_pos = mouse_vec # 스틱이 범위를 벗어나지 않도록 제한 if self.base_pos.distance_to(self.stick_pos) > self.radius: direction = (self.stick_pos - self.base_pos).normalize() self.stick_pos = self.base_pos + direction * self.radius else: # 터치를 떼면 원위치 self.active = False self.stick_pos = self.base_pos def get_movement(self): if self.active: vec = self.stick_pos - self.base_pos if vec.length() > 0: return vec / self.radius # 0.0 ~ 1.0 사이의 벡터값 반환 return pygame.math.Vector2(0, 0) def draw(self, surface): # 조이스틱 배경 (반투명 느낌 구현을 위해 테두리만) pygame.draw.circle(surface, (150, 150, 150), (int(self.base_pos.x), int(self.base_pos.y)), self.radius, 2) # 조이스틱 스틱 pygame.draw.circle(surface, (200, 200, 200), (int(self.stick_pos.x), int(self.stick_pos.y)), int(self.radius * 0.5)) class GameEngine: def __init__(self): # 모바일 비율에 맞춘 기차 위치 (화면 중앙 하단) self.train_rect = pygame.Rect(120, 250, 160, 350) self.control_rect = pygame.Rect(120, 250, 160, 50) # 플레이어 위치 self.player_x, self.player_y = 200, 450 self.player_speed = 4.0 # 기차 상태 self.is_moving = False self.train_speed = 0.0 self.max_speed = 8.0 self.inertia = 0.05 self.acceleration = 0.1 # 자원 self.food = 10 self.water = 10 self.fuel = 20.0 # 시각 효과 및 정거장 self.scroll_y = 0.0 self.last_tick_time = time.time() self.station = None self.scenery = [] for _ in range(15): self.spawn_tree(random.randint(0, HEIGHT)) # 모바일 UI 초기화 self.joystick = VirtualJoystick(80, 580, 50) # 좌측 하단 self.action_btn_rect = pygame.Rect(WIDTH - 120, 540, 90, 90) # 우측 하단 def spawn_tree(self, start_y): # 철로 좌우 풍경 x = random.randint(0, 80) if random.random() < 0.5 else random.randint(300, WIDTH) self.scenery.append(pygame.Rect(x, start_y, 30, 30)) def update(self, mouse_pos, mouse_pressed, mouse_just_pressed): # 1. 조이스틱 업데이트 및 플레이어 이동 self.joystick.update(mouse_pos, mouse_pressed) move_vec = self.joystick.get_movement() new_x = self.player_x + move_vec.x * self.player_speed new_y = self.player_y + move_vec.y * self.player_speed # 기차 내부 충돌 처리 if self.train_rect.collidepoint(new_x, self.player_y): self.player_x = new_x if self.train_rect.collidepoint(self.player_x, new_y): self.player_y = new_y # 2. 액션 버튼 터치 처리 (기차 조작부에 있을 때만) player_rect = pygame.Rect(self.player_x - 12, self.player_y - 12, 24, 24) is_near_control = player_rect.colliderect(self.control_rect) if is_near_control and mouse_just_pressed: if self.action_btn_rect.collidepoint(mouse_pos): self.is_moving = not self.is_moving # 3. 기차 속도 및 연료 관리 if self.is_moving: if self.train_speed < self.max_speed: self.train_speed += self.acceleration if self.fuel > 0: self.fuel -= 0.02 else: self.fuel = 0 self.is_moving = False else: if self.train_speed > 0: self.train_speed -= self.inertia if self.train_speed < 0: self.train_speed = 0 # 4. 배경 스크롤 및 정거장 이동 if self.train_speed > 0: self.scroll_y += self.train_speed for tree in self.scenery: tree.y += self.train_speed self.scenery = [t for t in self.scenery if t.y < HEIGHT + 50] while len(self.scenery) < 15: self.spawn_tree(random.randint(-50, -10)) if self.station: self.station["rect"].y += self.train_speed if self.station["rect"].y > HEIGHT + 100: self.station = None # 5. 정거장 생성 확률 로직 current_time = time.time() if current_time - self.last_tick_time >= 5.0 and self.is_moving: self.last_tick_time = current_time if self.station is None: roll_small = random.random() < 0.30 roll_large = random.random() < 0.05 if roll_small and roll_large: roll_large = False # 철로 옆(x=290)에 생성 if roll_small: self.station = {"name": "소규모", "rect": pygame.Rect(290, -150, 100, 100)} elif roll_large: self.station = {"name": "대규모", "rect": pygame.Rect(290, -200, 140, 140)} def draw(self): screen.fill(BG_COLOR) # 철로 (모바일 비율 중앙) pygame.draw.rect(screen, TRACK_COLOR, (140, 0, 10, HEIGHT)) pygame.draw.rect(screen, TRACK_COLOR, (250, 0, 10, HEIGHT)) tie_spacing = 40 offset = int(self.scroll_y) % tie_spacing for y in range(-tie_spacing + offset, HEIGHT, tie_spacing): pygame.draw.rect(screen, TIE_COLOR, (130, y, 140, 10)) # 배경 나무 for tree in self.scenery: pygame.draw.ellipse(screen, (20, 100, 20), tree) # 정거장 if self.station: pygame.draw.rect(screen, STATION_COLOR, self.station["rect"]) st_text = font.render(self.station["name"], True, BLACK) screen.blit(st_text, (self.station["rect"].x, self.station["rect"].y - 25)) # 기차 및 플레이어 pygame.draw.rect(screen, TRAIN_COLOR, self.train_rect) pygame.draw.rect(screen, CONTROL_PANEL_COLOR, self.control_rect) pygame.draw.circle(screen, PLAYER_COLOR, (int(self.player_x), int(self.player_y)), 12) # 상단 UI (자원) ui_food = font.render(f"🍔: {self.food}", True, WHITE) ui_water = font.render(f"💧: {self.water}", True, WHITE) ui_fuel = font.render(f"⛽: {self.fuel:.1f}", True, WHITE) screen.blit(ui_food, (10, 10)) screen.blit(ui_water, (70, 10)) screen.blit(ui_fuel, (130, 10)) # 모바일 터치 UI 그리기 self.joystick.draw(screen) # 플레이어가 조작부에 있을 때만 우측 하단에 '출발/정지' 버튼 표시 player_rect = pygame.Rect(self.player_x - 12, self.player_y - 12, 24, 24) if player_rect.colliderect(self.control_rect): btn_color = (255, 100, 100) if self.is_moving else (100, 255, 100) pygame.draw.rect(screen, btn_color, self.action_btn_rect, border_radius=15) btn_text = font.render("정지" if self.is_moving else "출발", True, BLACK) text_rect = btn_text.get_rect(center=self.action_btn_rect.center) screen.blit(btn_text, text_rect) pygame.display.flip() def main(): engine = GameEngine() clock = pygame.time.Clock() running = True # 마우스 클릭(터치) 시작 이벤트를 감지하기 위한 변수 mouse_was_pressed = False while running: mouse_pos = pygame.mouse.get_pos() mouse_pressed = pygame.mouse.get_pressed() # 방금 터치(클릭)된 순간인지 확인 mouse_just_pressed = False if mouse_pressed[0] and not mouse_was_pressed: mouse_just_pressed = True mouse_was_pressed = mouse_pressed[0] for event in pygame.event.get(): if event.type == pygame.QUIT: running = False engine.update(mouse_pos, mouse_pressed, mouse_just_pressed) engine.draw() clock.tick(60) pygame.quit() if __name__ == "__main__": main()
